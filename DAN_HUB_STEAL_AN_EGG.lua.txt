local function NeutralizeTraps()
	local debris = Workspace:FindFirstChild("__DEBRIS")
	if not debris then return end
	for _, d in ipairs(debris:GetChildren()) do
		if d.Name == "PlayerTrap" and d:GetAttribute("Owner") ~= LP.Name then
			if d:IsA("BasePart") then
				d.CanTouch = false
				d.CanQuery = false
			end
			for _, c in ipairs(d:GetChildren()) do
				if c:IsA("BasePart") then
					c.CanTouch = false
					c.CanQuery = false
					if c.Name == "Hitbox" then
						c.CFrame = CFrame.new(0, -999, 0)
					end
				end
			end
		end
	end
end

local function MoveToPoint(target, speed)
	local hrp = findHRP()
	if not hrp or not target then return false end
	speed = math.clamp(tonumber(speed) or 500, 50, 750)
	local t0 = os.clock()
	local totalDist = (target - hrp.Position).Magnitude
	while not dead do
		RunService.Heartbeat:Wait()
		local remain = (target - hrp.Position).Magnitude
		if remain < 1.5 then break end
		local dir = (target - hrp.Position).Unit
		local step = math.min(speed * 0.03, remain)
		local nextPos = hrp.Position + dir * step
		hrp.CFrame = CFrame.lookAt(nextPos, nextPos + dir)
		hrp.AssemblyLinearVelocity = Vector3.zero
		if os.clock() - t0 > (totalDist / 40 + 8) then break end
	end
	hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
	hrp.AssemblyLinearVelocity = Vector3.zero
	return true
end

local function TravelRoadPath(targetPos, speed)
	local hrp = findHRP()
	if not hrp or not targetPos then return false end
	if avoidTrapsEnabled then pcall(NeutralizeTraps) end
	local startPos = hrp.Position
	local safeY = math.max(startPos.Y, targetPos.Y, 70.4)
	if targetPos.X < 560 and startPos.X > SAFE_BOUNDARY_X then
		MoveToPoint(Vector3.new(startPos.X, safeY, MAIN_ROAD_Z), speed)
		MoveToPoint(Vector3.new(SAFE_BOUNDARY_X, safeY, MAIN_ROAD_Z), speed)
		MoveToPoint(Vector3.new(targetPos.X, safeY, MAIN_ROAD_Z), SAFE_ZONE_SPEED)
		MoveToPoint(targetPos + Vector3.new(0, 1.2, 0), SAFE_ZONE_SPEED)
	else
		MoveToPoint(Vector3.new(startPos.X, safeY, MAIN_ROAD_Z), speed)
		MoveToPoint(Vector3.new(targetPos.X, safeY, MAIN_ROAD_Z), speed)
		MoveToPoint(targetPos + Vector3.new(0, 1.2, 0), speed)
	end
	return true
end

local function TravelFlyDirect(targetPos, speed)
	local hrp = findHRP()
	if not hrp or not targetPos then return false end
	if avoidTrapsEnabled then pcall(NeutralizeTraps) end
	local start = hrp.Position
	local alt = math.max(start.Y, targetPos.Y, 70.4) + 28
	MoveToPoint(Vector3.new(start.X, alt, start.Z), speed)
	MoveToPoint(Vector3.new(targetPos.X, alt, targetPos.Z), speed)
	MoveToPoint(Vector3.new(targetPos.X, math.max(targetPos.Y, 70) + 1.2, targetPos.Z), speed)
	return true
end

local function TravelToDestination(targetPos, speed)
	if stealMovementMethod == "Fly Glide" then
		return TravelFlyDirect(targetPos, speed)
	elseif stealMovementMethod == "Teleport" then
		local hrp = findHRP()
		if hrp then
			hrp.CFrame = CFrame.new(targetPos + Vector3.new(0, 1.5, 0))
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
		return true
	end
	return TravelRoadPath(targetPos, speed)
end

local function GetEggRarityInfo(egg)
	if not egg then return "Common", 100 end
	if egg.Rarity then
		local r = egg.Rarity
		local name = type(r) == "table" and (r.DisplayName or r._id or r.Name) or tostring(r)
		return name, RARITY_SCORE_MAP[name] or 100
	end
	local cat = egg.AssetCategory or egg.Category or egg.Name
	if cat and AssetsData then
		local aInfo = (AssetsData.Directory or AssetsData)[cat]
		if aInfo and aInfo.Rarity then
			local r = aInfo.Rarity
			local name = type(r) == "table" and (r.DisplayName or r._id or r.Name) or tostring(r)
			return name, RARITY_SCORE_MAP[name] or 100
		end
	end
	return "Common", 100
end

local function filterHas(filter)
	if type(filter) ~= "table" then return false end
	for _ in pairs(filter) do return true end
	return false
end

local function isRarityAllowed(name, filter)
	if not filterHas(filter) then return true end
	local rLower = string.lower(tostring(name))
	for k, v in pairs(filter) do
		if type(v) == "string" and string.lower(v) == rLower then return true end
		if type(k) == "string" and string.lower(k) == rLower and v then return true end
	end
	return false
end

local function isAreaAllowed(areaId, filter)
	if not filterHas(filter) then return true end
	local aLower = string.lower(tostring(areaId))
	for k, v in pairs(filter) do
		if type(v) == "string" and string.lower(v) == aLower then return true end
		if type(k) == "string" and string.lower(k) == aLower and v then return true end
	end
	return false
end

local function GetMatchingFieldEggs()
	if not EggState or not EggState.ReadFieldEggs then return {} end
	local ok, snapshot = pcall(EggState.ReadFieldEggs)
	if not ok or not snapshot or not snapshot.Records then return {} end
	local matched = {}
	for _, record in ipairs(snapshot.Records) do
		if record.State == "Slot" and record.BoundsCFrame then
			local isIgnored = ignoredEggs[record.Uid] and (os.clock() - ignoredEggs[record.Uid] < 2.5)
			if not isIgnored then
				local rarityName, baseScore = GetEggRarityInfo(record)
				if isAreaAllowed(record.AreaId, selectedStealAreas) and isRarityAllowed(rarityName, selectedStealRarities) then
					local mutBonus = 0
					for _, m in ipairs(record.Mutations or {}) do
						if m == "Rainbow" then mutBonus = mutBonus + 35 end
						if m == "Gold" or m == "Golden" then mutBonus = mutBonus + 20 end
					end
					if record.HasParasite then mutBonus = mutBonus + 800 end
					matched[#matched + 1] = { record = record, rarity = rarityName, score = baseScore + mutBonus }
				end
			end
		end
	end
	if rareEggHunter and #matched > 1 then
		table.sort(matched, function(a, b) return a.score > b.score end)
	end
	return matched
end

local function isPlayerCarryingEgg()
	local pg = LP:FindFirstChildOfClass("PlayerGui")
	local dropGui = pg and pg:FindFirstChild("DropHeldEgg")
	if dropGui and dropGui.Enabled then return true end
	local char = LP.Character
	if char then
		for _, t in ipairs(char:GetChildren()) do
			if t:IsA("Tool") then
				if EggToolDisplay and EggToolDisplay.IsEggTool and EggToolDisplay.IsEggTool(t) then return true end
				if t:GetAttribute("IsEgg") or t:GetAttribute("Uid") or t:GetAttribute("AssetCategory") then return true end
			end
		end
	end
	return false
end

local function PlantAllCarriedEggsInPen()
	local toolsToPlant = {}
	local function collect(container)
		if not container then return end
		for _, t in ipairs(container:GetChildren()) do
			if t:IsA("Tool") and EggToolDisplay and EggToolDisplay.IsEggTool and EggToolDisplay.IsEggTool(t) then
				local uid = EggToolDisplay.GetToolUid and EggToolDisplay.GetToolUid(t)
				if uid then toolsToPlant[#toolsToPlant + 1] = uid end
			end
		end
	end
	collect(LP.Character)
	collect(LP:FindFirstChild("Backpack"))
	local planted = 0
	for _, eggUid in ipairs(toolsToPlant) do
		for _ = 1, 3 do
			local offset = CFrame.new(math.random(-6, 6), 0, math.random(-6, 6))
			local ok, res = pcall(function()
				if EggState and EggState.PlantEgg then
					return EggState.PlantEgg(eggUid, offset)
				end
			end)
			if ok and res then
				planted = planted + 1
				break
			end
			task.wait(0.08)
		end
	end
	return planted
end

local function StealSpecificEggRobust(targetItem)
	local record = targetItem.record or targetItem
	if not record or not record.Uid or not record.BoundsCFrame then return false end
	local hrp = findHRP()
	if not hrp then return false end

	local targetPos = record.BoundsCFrame.Position
	local speed = math.clamp(tonumber(glideSpeed) or 500, 50, 750)
	TravelToDestination(targetPos + Vector3.new(0, 1.2, 0), speed)
	task.wait(0.35)

	local slotKey = nil
	pcall(function()
		if AreaEggSlotIdentity and AreaEggSlotIdentity.LooksLikeFirstAreaUid and AreaEggSlotIdentity.LooksLikeFirstAreaUid(record.Uid) then
			slotKey = AreaEggSlotIdentity.SlotKey(record.AreaId, record.NestId)
		end
	end)

	local carryRemote = GetNetRemote("RF/EggWorld/AskFieldEggCarry")
	if carryRemote then
		pcall(function() carryRemote:InvokeServer({ Uid = record.Uid, FirstAreaSlotKey = slotKey }) end)
	end
	pcall(function()
		if EggState and EggState.CarryFieldEgg then
			EggState.CarryFieldEgg(record.Uid, slotKey)
		end
	end)

	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Name == "CarryAreaEgg" and d.Enabled then
			local p = d.Parent
			if p:IsA("Attachment") then p = p.Parent end
			if p and p:IsA("BasePart") and (p.Position - hrp.Position).Magnitude < 16 then
				d.HoldDuration = 0
				pcall(function() fireproximityprompt(d) end)
				break
			end
		end
	end

	local t0 = os.clock()
	local carried = false
	while os.clock() - t0 < 1.5 and not dead do
		if isPlayerCarryingEgg() then
			carried = true
			break
		end
		pcall(function()
			if EggState and EggState.CarryFieldEgg then
				EggState.CarryFieldEgg(record.Uid, slotKey)
			end
		end)
		task.wait(0.05)
	end

	local plot = GetLocalPlotCenter()
	TravelToDestination(plot + Vector3.new(0, 1.2, 0), speed)
	if autoPlantEnabled or carried then
		task.wait(0.2)
		PlantAllCarriedEggsInPen()
	end
	if not carried then
		ignoredEggs[record.Uid] = os.clock()
	end
	return carried or isPlayerCarryingEgg()
end

local function HatchAllReadyEggs()
	if not EggState or not EggState.ReadOwnedEggs then return 0 end
	local ok, snapshot = pcall(EggState.ReadOwnedEggs, LP.UserId)
	if not ok or not snapshot then return 0 end
	local count = 0
	local records = snapshot.Records or snapshot
	if type(records) == "table" then
		for uid, eggData in pairs(records) do
			if type(eggData) == "table" then
				local ready = false
				if EggState.IsReadyToHatch then
					ready = EggState.IsReadyToHatch(eggData)
				else
					ready = eggData.Placement ~= nil
				end
				if ready then
					pcall(function()
						if EggState.BeginHatch then EggState.BeginHatch(uid) end
						task.wait(0.05)
						if EggState.FinishHatch then EggState.FinishHatch(uid) end
						count = count + 1
					end)
				end
			end
		end
	end
	return count
end

local function StealBestEggOnce()
	pcall(HatchAllReadyEggs)
	local eggs = GetMatchingFieldEggs()
	if #eggs == 0 then return false end
	return StealSpecificEggRobust(eggs[1])
end

local function SellSelectedPets()
	local re = GetNetRemote("RE/PetSatchel/SellPet")
	if not re or not SaveModule then return end
	local save
	pcall(function() save = SaveModule.Get and SaveModule.Get() end)
	local inv = save and save.Inventory
	if type(inv) ~= "table" then return end
	local low = { Common = true, Uncommon = true, Rare = true, Epic = true, Legendary = true, Mythic = true }
	for uid, petData in pairs(inv) do
		if type(petData) == "table" and not petData.Locked then
			local rName = petData.Rarity or "Common"
			local okSell = false
			if filterHas(selectedStealRarities) then
				okSell = isRarityAllowed(rName, selectedStealRarities)
			else
				okSell = low[rName] == true
			end
			if okSell then
				pcall(function() re:FireServer(uid) end)
				task.wait(0.08)
			end
		end
	end
end
